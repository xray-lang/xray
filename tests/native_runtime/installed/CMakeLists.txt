cmake_minimum_required(VERSION 3.25)
project(NativeRuntimeClients LANGUAGES C CXX)
if(TEST_NO_VERSION)
    find_package(XrayNativeRuntime CONFIG REQUIRED)
else()
    find_package(XrayNativeRuntime 1.0.0 EXACT CONFIG REQUIRED)
endif()
if(DEFINED TEST_SECOND_PREFIX)
    unset(XrayNativeRuntime_DIR CACHE)
    find_package(XrayNativeRuntime 1.0.0 EXACT CONFIG REQUIRED
        PATHS "${TEST_SECOND_PREFIX}/lib/cmake/XrayNativeRuntime" NO_DEFAULT_PATH)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/cases.cmake")
foreach(client IN LISTS CLIENTS)
    add_executable(${client} ${${client}_SOURCES})
    target_link_libraries(${client} PRIVATE Xray::NativeRuntime)
    set_target_properties(${client} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON
        C_EXTENSIONS OFF CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    target_compile_options(${client} PRIVATE /W4 /WX /utf-8 /GL-)
    if(DEFINED TEST_CRT_OVERRIDE)
        set_property(TARGET ${client} PROPERTY MSVC_RUNTIME_LIBRARY "${TEST_CRT_OVERRIDE}")
    endif()
    if(DEFINED TEST_EXTRA_OPTIONS)
        target_compile_options(${client} PRIVATE ${TEST_EXTRA_OPTIONS})
    endif()
endforeach()
