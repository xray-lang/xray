cmake_minimum_required(VERSION 3.21)
project(xray_toml_owner_tests LANGUAGES C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "TOML owner tests require Ninja")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(test_xtoml "${CMAKE_CURRENT_LIST_DIR}/../test_xtoml.c"
    "${XR_ROOT}/src/base/xtoml.c" "${XR_ROOT}/src/base/xio_policy.c"
    "${XR_ROOT}/src/base/xcompile_resources.c")
target_include_directories(test_xtoml PRIVATE "${XR_ROOT}/src")
target_compile_definitions(test_xtoml PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(test_xtoml PRIVATE /utf-8 /W4 /WX)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(test_xtoml PRIVATE /experimental:c11atomics)
    endif()
endif()
enable_testing()
add_test(NAME test_xtoml COMMAND test_xtoml)
set_tests_properties(test_xtoml PROPERTIES LABELS "unit;metadata;ownership;budget" TIMEOUT 120)
add_executable(test_xtoml_allocations "${CMAKE_CURRENT_LIST_DIR}/../test_xtoml_allocations.c"
    "${XR_ROOT}/src/base/xio_policy.c")
target_include_directories(test_xtoml_allocations PRIVATE "${XR_ROOT}/src")
target_compile_definitions(test_xtoml_allocations PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(test_xtoml_allocations PRIVATE /utf-8 /W4 /WX)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(test_xtoml_allocations PRIVATE /experimental:c11atomics)
    endif()
endif()
add_test(NAME test_xtoml_allocations COMMAND test_xtoml_allocations)
set_tests_properties(test_xtoml_allocations PROPERTIES LABELS "unit;metadata;ownership;budget" TIMEOUT 120)
