cmake_minimum_required(VERSION 3.21)
project(xray_file_read_unix_tests LANGUAGES C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "File input tests require Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
add_compile_definitions(_POSIX_C_SOURCE=200809L)
if(NOT TARGET xray_compile_resources)
    add_subdirectory("${XR_ROOT}/src/base" compiler-resources EXCLUDE_FROM_ALL)
endif()
add_executable(test_file_read_unix ../test_file_read_unix.c
    "${XR_ROOT}/src/base/xio_policy.c" "${XR_ROOT}/src/os/unix/file_read_unix.c")
target_link_libraries(test_file_read_unix PRIVATE xray_compile_resources)
target_include_directories(test_file_read_unix PRIVATE "${XR_ROOT}/src")
target_compile_features(test_file_read_unix PRIVATE c_std_11)
set_target_properties(test_file_read_unix PROPERTIES C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_compile_options(test_file_read_unix PRIVATE -Wall -Wextra -Werror -pedantic)
enable_testing()
add_test(NAME test_file_read_unix COMMAND ${Python3_EXECUTABLE}
    "${XR_ROOT}/tests/unit/base/test_file_read_unix.py" $<TARGET_FILE:test_file_read_unix>)
set_tests_properties(test_file_read_unix PROPERTIES LABELS "unit;metadata;ownership;filesystem;posix" TIMEOUT 30)
