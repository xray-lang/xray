cmake_minimum_required(VERSION 3.21)
project(xray_file_read_owner_tests LANGUAGES C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "File owner tests require Ninja")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_executable(test_file_read_owner ../test_file_read_windows.c
    "${XR_ROOT}/src/base/xio_policy.c" "${XR_ROOT}/src/os/win/file_read_win.c")
target_include_directories(test_file_read_owner PRIVATE "${XR_ROOT}/src")
target_compile_definitions(test_file_read_owner PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(test_file_read_owner PRIVATE /utf-8 /W4 /WX)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(test_file_read_owner PRIVATE /experimental:c11atomics)
    endif()
endif()
add_test(NAME test_file_read_owner COMMAND ${Python3_EXECUTABLE}
    "${XR_ROOT}/tests/unit/base/test_file_read_windows.py" $<TARGET_FILE:test_file_read_owner>)
set_tests_properties(test_file_read_owner PROPERTIES LABELS "unit;metadata;ownership;windows;budget" TIMEOUT 120)
