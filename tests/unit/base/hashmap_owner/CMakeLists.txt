cmake_minimum_required(VERSION 3.21)
project(xray_hashmap_owner_tests LANGUAGES C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Hashmap owner tests require Ninja")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(test_hashmap_owner test_hashmap_owner.c "${XR_ROOT}/src/base/xhashmap.c" "${XR_ROOT}/src/base/xio_policy.c")
target_include_directories(test_hashmap_owner PRIVATE "${XR_ROOT}/src")
target_compile_definitions(test_hashmap_owner PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(test_hashmap_owner PRIVATE /utf-8 /W4 /WX)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(test_hashmap_owner PRIVATE /experimental:c11atomics)
    endif()
endif()
enable_testing()
add_test(NAME test_hashmap_owner COMMAND test_hashmap_owner)
set_tests_properties(test_hashmap_owner PROPERTIES LABELS "unit;metadata;ownership;budget" TIMEOUT 120)
