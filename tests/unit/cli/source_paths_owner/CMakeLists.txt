cmake_minimum_required(VERSION 3.25)
project(XirSourcePaths C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Source path qualification requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SOURCE_PATHS_SANITIZE "Enable Clang ASan and UBSan" OFF)
if(SOURCE_PATHS_SANITIZE)
    execute_process(COMMAND "${CMAKE_C_COMPILER}" /clang:-print-resource-dir
        OUTPUT_VARIABLE resource OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
endif()
foreach(mode injected production)
    add_executable(test_source_paths_${mode} test_source_paths.c
        "${XR_ROOT}/src/base/xio_policy.c" "${XR_ROOT}/src/base/xfileio.c"
        "${XR_ROOT}/src/os/win/fs_win.c" "${XR_ROOT}/src/app/cli/xcli_source_paths.c")
    if(mode STREQUAL "production")
        target_compile_definitions(test_source_paths_${mode} PRIVATE SOURCE_PATHS_PRODUCTION)
        target_sources(test_source_paths_${mode} PRIVATE "${XR_ROOT}/src/os/win/proc_self_win.c")
    endif()
    target_include_directories(test_source_paths_${mode} PRIVATE "${XR_ROOT}/src" "${XR_ROOT}/include")
    target_compile_definitions(test_source_paths_${mode} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(test_source_paths_${mode} PRIVATE /W4 /WX /utf-8)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(test_source_paths_${mode} PRIVATE /experimental:c11atomics)
    endif()
    if(SOURCE_PATHS_SANITIZE)
        target_compile_options(test_source_paths_${mode} PRIVATE /fsanitize=address /Oy-
            /clang:-fsanitize=undefined /clang:-fno-sanitize=function)
        target_link_options(test_source_paths_${mode} PRIVATE /INCREMENTAL:NO /INCLUDE:__asan_seh_interceptor
            "/WHOLEARCHIVE:${resource}/lib/windows/clang_rt.asan_dynamic_runtime_thunk-x86_64.lib")
        target_link_libraries(test_source_paths_${mode} PRIVATE "${resource}/lib/windows/clang_rt.asan_dynamic-x86_64.lib")
    endif()
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
foreach(mode injected production)
    add_test(NAME source_paths_${mode} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_source_paths.py" $<TARGET_FILE:test_source_paths_${mode}>)
    set_tests_properties(source_paths_${mode} PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE LABELS "unit;cli;ownership;budget")
    if(SOURCE_PATHS_SANITIZE)
        set_tests_properties(source_paths_${mode} PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${resource}/lib/windows")
    endif()
endforeach()
