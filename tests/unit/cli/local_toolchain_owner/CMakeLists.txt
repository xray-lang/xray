cmake_minimum_required(VERSION 3.25)
project(LocalToolchainOwner C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This gate requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT IS_ABSOLUTE "${LOCAL_SOURCE_BUILD}" OR NOT IS_ABSOLUTE "${LOCAL_SDK}")
    message(FATAL_ERROR "Provide qualified LOCAL_SOURCE_BUILD and LOCAL_SDK")
endif()
option(LOCAL_SANITIZE "Instrument the new owner and test" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(/W4 /WX /utf-8)
add_compile_definitions(NDEBUG WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    add_compile_options(/experimental:c11atomics)
endif()
include_directories("${XR_ROOT}/src" "${XR_ROOT}/include")
set(local_libraries)
foreach(item xray_toolchain_process xray_xir_target src/toolchain/xray_xir_runtime_sdk src/base/xray_json_cursor)
    set(path "${LOCAL_SOURCE_BUILD}/${item}.lib")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Missing actual production archive: ${path}")
    endif()
    list(APPEND local_libraries "${path}")
endforeach()
add_library(local_dependencies STATIC "${XR_ROOT}/src/base/xio_policy.c"
    "${XR_ROOT}/src/base/xfileio.c" "${XR_ROOT}/src/os/win/fs_win.c")
target_link_libraries(local_dependencies PUBLIC ${local_libraries})
add_library(local_production STATIC "${XR_ROOT}/src/app/toolchain/xtc_xir_local_toolchain.c")
target_link_libraries(local_production PUBLIC local_dependencies)
foreach(mode production injected)
    add_executable(test_local_toolchain_${mode} test_local_toolchain.c)
    if(mode STREQUAL "production")
        target_compile_definitions(test_local_toolchain_${mode} PRIVATE LOCAL_PRODUCTION)
        target_link_libraries(test_local_toolchain_${mode} PRIVATE local_production)
    else()
        target_link_libraries(test_local_toolchain_${mode} PRIVATE local_dependencies)
    endif()
endforeach()
if(LOCAL_SANITIZE)
    execute_process(COMMAND "${CMAKE_C_COMPILER}" /clang:-print-resource-dir
        OUTPUT_VARIABLE resource OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    foreach(target local_dependencies local_production test_local_toolchain_production test_local_toolchain_injected)
        target_compile_options(${target} PRIVATE /fsanitize=address /Oy-
            /clang:-fsanitize=undefined /clang:-fno-sanitize=function)
    endforeach()
    foreach(target test_local_toolchain_production test_local_toolchain_injected)
        target_link_options(${target} PRIVATE /INCREMENTAL:NO /INCLUDE:__asan_seh_interceptor
            "/WHOLEARCHIVE:${resource}/lib/windows/clang_rt.asan_dynamic_runtime_thunk-x86_64.lib")
        target_link_libraries(${target} PRIVATE "${resource}/lib/windows/clang_rt.asan_dynamic-x86_64.lib")
    endforeach()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_program(LOCAL_MSVC NAMES cl REQUIRED)
enable_testing()
add_test(NAME local_toolchain_faults COMMAND test_local_toolchain_injected)
add_test(NAME local_toolchain_real COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${CMAKE_CURRENT_LIST_DIR}/test_local_toolchain.py" "$<TARGET_FILE:test_local_toolchain_production>"
    "${LOCAL_SDK}" "${LOCAL_MSVC}" "${CMAKE_CURRENT_BINARY_DIR}/cases")
set_tests_properties(local_toolchain_faults local_toolchain_real PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
if(LOCAL_SANITIZE)
    set_tests_properties(local_toolchain_faults local_toolchain_real PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${resource}/lib/windows")
endif()
