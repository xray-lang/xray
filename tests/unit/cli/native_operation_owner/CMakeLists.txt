cmake_minimum_required(VERSION 3.25)
project(NativeOperationOwner C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This gate requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT IS_ABSOLUTE "${OPERATION_SOURCE_BUILD}" OR NOT IS_ABSOLUTE "${OPERATION_SDK}")
    message(FATAL_ERROR "Provide qualified OPERATION_SOURCE_BUILD and OPERATION_SDK")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(/W4 /WX /utf-8)
add_compile_definitions(NDEBUG WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    add_compile_options(/experimental:c11atomics)
endif()
include_directories("${XR_ROOT}/src" "${XR_ROOT}/include" "${OPERATION_SDK}")
set(operation_libraries)
foreach(item src/xir/xray_xir_source_product src/xir/xray_xir_source src/xir/xray_xir_vm
    src/xir/xray_xir_scalar src/xir/xray_xir_cgen src/xir/xray_xir src/xir/xray_xir_admission
    src/xir/xray_xir_declarations src/xir/xray_compiler_parser src/base/xray_compile_resources
    src/toolchain/xray_xir_native_projection src/toolchain/xray_xir_native_artifact
    src/toolchain/xray_toolchain_binding src/toolchain/xray_xir_runtime_sdk
    src/base/xray_json_cursor xray_xir_target xray_toolchain_dependencies xray_toolchain_process
    xray_xir_workspace xray_xir_pe_version)
    get_filename_component(name "${item}" NAME)
    set(path "${OPERATION_SOURCE_BUILD}/${item}.lib")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Missing qualified archive: ${path}")
    endif()
    add_library(${name} STATIC IMPORTED)
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${path}")
    list(APPEND operation_libraries ${name})
endforeach()
# Compile the current composition and invocation against qualified dependencies.
add_library(operation_dependencies STATIC
    "${XR_ROOT}/src/app/toolchain/xtc_xir_invocation.c")
target_link_libraries(operation_dependencies PUBLIC ${operation_libraries} ntdll bcrypt)
add_library(operation_production STATIC "${XR_ROOT}/src/app/toolchain/xtc_xir_native_operation.c")
target_link_libraries(operation_production PUBLIC operation_dependencies)
add_executable(test_native_operation_owner test_native_operation_owner.c)
target_link_libraries(test_native_operation_owner PRIVATE operation_production)
add_executable(test_native_operation_faults test_native_operation_owner.c)
target_compile_definitions(test_native_operation_faults PRIVATE OPERATION_INJECTED)
target_link_libraries(test_native_operation_faults PRIVATE operation_dependencies)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_program(OPERATION_MSVC NAMES cl REQUIRED)
find_program(OPERATION_LINKER NAMES link REQUIRED)
enable_testing()
foreach(mode unit preflight native)
    if(mode STREQUAL "native")
        set(executable test_native_operation_owner)
    else()
        set(executable test_native_operation_faults)
    endif()
    add_test(NAME native_operation_${mode} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_native_operation_owner.py" $<TARGET_FILE:${executable}>
        "${XR_ROOT}" "${OPERATION_MSVC}" "${OPERATION_LINKER}" "${OPERATION_SDK}" ${mode})
    set_tests_properties(native_operation_${mode} PROPERTIES TIMEOUT 300 RUN_SERIAL TRUE)
endforeach()
