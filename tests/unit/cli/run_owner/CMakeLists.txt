cmake_minimum_required(VERSION 3.25)
project(XirRunCommandOwner C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "The run command gate requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
foreach(directory RUN_SOURCE_BUILD)
    if(NOT IS_ABSOLUTE "${${directory}}")
        message(FATAL_ERROR "Set ${directory} to the actual fresh build directory")
    endif()
endforeach()
set(run_libraries "${RUN_SOURCE_BUILD}/xray_cli_source.lib"
    "${RUN_SOURCE_BUILD}/src/execution/xray_xir_runtime_host.lib"
    "${RUN_SOURCE_BUILD}/src/base/xray_compile_resources.lib")
foreach(name xray_xir_source_product xray_xir_source xray_xir_vm xray_xir_cgen xray_xir xray_compiler_parser xray_xir_scalar xray_xir_admission xray_xir_declarations)
    list(APPEND run_libraries "${RUN_SOURCE_BUILD}/src/xir/${name}.lib")
endforeach()
foreach(library IN LISTS run_libraries)
    if(NOT EXISTS "${library}")
        message(FATAL_ERROR "Missing production archive: ${library}")
    endif()
endforeach()
set(cli_sources "${XR_ROOT}/src/app/cli/xcli_diag.c")
add_executable(run_owner_driver run_owner_driver.c "${XR_ROOT}/src/app/cli/xcmd_run.c" ${cli_sources})
add_executable(run_owner_injected run_owner_driver.c ${cli_sources})
add_executable(run_owner_emit "${XR_ROOT}/tests/unit/xir/host_cli_owner/emit_host_cli.c")
target_compile_definitions(run_owner_injected PRIVATE RUN_OWNER_INJECTED=1)
foreach(target run_owner_driver run_owner_injected run_owner_emit)
    target_include_directories(${target} PRIVATE "${XR_ROOT}/src" "${XR_ROOT}/include")
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    target_link_libraries(${target} PRIVATE ${run_libraries} bcrypt)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(${target} PRIVATE /experimental:c11atomics)
    endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME run_source_owner COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${CMAKE_CURRENT_LIST_DIR}/run_owner_cases.py" --root "${XR_ROOT}"
    --driver "$<TARGET_FILE:run_owner_driver>" --injected "$<TARGET_FILE:run_owner_injected>"
    --output "${CMAKE_CURRENT_BINARY_DIR}/cases")
set_tests_properties(run_source_owner PROPERTIES TIMEOUT 120 COST 2)
add_test(NAME run_source_generated COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${XR_ROOT}/tests/unit/xir/host_cli_owner/source_cases.py" --root "${XR_ROOT}"
    --emitter "$<TARGET_FILE:run_owner_emit>" --output "${CMAKE_CURRENT_BINARY_DIR}/generated-cases")
set_tests_properties(run_source_generated PROPERTIES TIMEOUT 90 COST 1)
