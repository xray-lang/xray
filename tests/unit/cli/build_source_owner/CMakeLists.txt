cmake_minimum_required(VERSION 3.25)
project(BuildSourceOwner C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This gate requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT IS_ABSOLUTE "${BUILD_SOURCE_ARCHIVES}")
    message(FATAL_ERROR "Provide the fresh production archive directory")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(/W4 /WX /utf-8)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    add_compile_options(/experimental:c11atomics)
endif()
include_directories("${XR_ROOT}/src" "${XR_ROOT}/include" "${BUILD_SOURCE_ARCHIVES}/generated")
set(build_libraries)
foreach(item xray_cli_source xray_xir_target src/base/xray_compile_resources
    src/toolchain/xray_xir_native_projection src/toolchain/xray_xir_native_artifact src/toolchain/xray_toolchain_binding
    src/xir/xray_xir_source_product src/xir/xray_xir_source src/xir/xray_xir_vm src/xir/xray_xir_cgen
    src/xir/xray_xir src/xir/xray_compiler_parser src/xir/xray_xir_scalar src/xir/xray_xir_admission src/xir/xray_xir_declarations)
    set(path "${BUILD_SOURCE_ARCHIVES}/${item}.lib")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Missing actual production archive: ${path}")
    endif()
    list(APPEND build_libraries "${path}")
endforeach()
add_library(publication_lease STATIC "${XR_ROOT}/src/app/toolchain/xtc_xir_sysroot.c")
target_link_libraries(publication_lease PUBLIC ${build_libraries})
add_library(publication_production STATIC "${XR_ROOT}/src/app/toolchain/xtc_xir_publication.c")
target_link_libraries(publication_production PUBLIC publication_lease bcrypt)
set(cli_sources "${XR_ROOT}/src/app/cli/xcmd_build.c" "${XR_ROOT}/src/app/cli/xcli_spec.c"
    "${XR_ROOT}/src/app/cli/xcli_parser.c" "${XR_ROOT}/src/app/cli/xcli_diag.c"
    "${XR_ROOT}/src/app/cli/xcli_output.c" "${XR_ROOT}/src/app/cli/xcli_help.c"
    "${XR_ROOT}/src/os/win/fd_win.c" "${XR_ROOT}/src/os/win/time_win.c")
add_executable(build_source_driver build_source_driver.c ${cli_sources})
target_link_libraries(build_source_driver PRIVATE publication_production)
add_executable(build_source_faults build_source_driver.c ${cli_sources})
target_compile_definitions(build_source_faults PRIVATE BUILD_SOURCE_INJECTED=1)
target_link_libraries(build_source_faults PRIVATE publication_lease bcrypt)
add_executable(test_publication_owner test_publication_owner.c)
target_link_libraries(test_publication_owner PRIVATE publication_production)
add_executable(test_publication_faults test_publication_owner.c)
target_compile_definitions(test_publication_faults PRIVATE PUBLICATION_INJECTED=1)
target_link_libraries(test_publication_faults PRIVATE publication_lease bcrypt)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME build_source_owner COMMAND "${Python3_EXECUTABLE}" -X utf8 "${CMAKE_CURRENT_LIST_DIR}/build_source_cases.py"
    --root "${XR_ROOT}" --driver "$<TARGET_FILE:build_source_driver>" --output "${CMAKE_CURRENT_BINARY_DIR}/cases"
    --publication "$<TARGET_FILE:test_publication_owner>" --faults "$<TARGET_FILE:test_publication_faults>"
    --fault-driver "$<TARGET_FILE:build_source_faults>")
set_tests_properties(build_source_owner PROPERTIES TIMEOUT 120 COST 3)
