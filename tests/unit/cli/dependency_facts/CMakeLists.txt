cmake_minimum_required(VERSION 3.25)
project(DependencyFacts C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This gate requires Ninja")
endif()
get_filename_component(DEPENDENCY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
foreach(mode injected production)
    add_executable(test_dependencies_${mode} test_dependency_facts.c
        "${DEPENDENCY_ROOT}/src/app/toolchain/xtc_dependencies.c"
        "${DEPENDENCY_ROOT}/src/base/xjson_cursor.c")
    target_include_directories(test_dependencies_${mode} PRIVATE "${DEPENDENCY_ROOT}/src")
    target_compile_definitions(test_dependencies_${mode} PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(mode STREQUAL "production")
        target_compile_definitions(test_dependencies_${mode} PRIVATE DEPENDENCY_PRODUCTION)
        target_sources(test_dependencies_${mode} PRIVATE "${DEPENDENCY_ROOT}/src/base/xcompile_resources.c")
    endif()
    if(MSVC)
        target_compile_options(test_dependencies_${mode} PRIVATE /W4 /WX /utf-8)
        if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
            target_compile_options(test_dependencies_${mode} PRIVATE /experimental:c11atomics)
        endif()
    else()
        target_compile_options(test_dependencies_${mode} PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME dependencies_${mode} COMMAND test_dependencies_${mode} "${CMAKE_CURRENT_LIST_DIR}/fixtures")
    set_tests_properties(dependencies_${mode} PROPERTIES TIMEOUT 180)
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME dependency_fixture_hashes COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_LIST_DIR}/verify_fixtures.py")
if(DEPENDENCY_SDK_REGRESSION)
    set(XIR_TARGET_SOURCE_ROOT "${DEPENDENCY_ROOT}")
    include("${CMAKE_CURRENT_LIST_DIR}/../../xir/sdk_resources.cmake")
    foreach(target xir_sdk_resources_production test_xir_sdk_resources test_xir_sdk_resources_lease)
        if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
            target_compile_options(${target} PRIVATE /experimental:c11atomics)
        endif()
    endforeach()
endif()
