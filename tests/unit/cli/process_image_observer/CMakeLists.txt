cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ProcessImageObserver C)
endif()
if(NOT CMAKE_GENERATOR STREQUAL "Ninja" OR NOT WIN32)
    message(FATAL_ERROR "This independent gate requires Windows Ninja")
endif()
get_filename_component(OBSERVER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_library(image_observer_dll SHARED image_observer_dll.c)
add_executable(test_process_image_observer test_process_image_observer.c
    "${OBSERVER_ROOT}/src/base/xutf8.c" "${OBSERVER_ROOT}/src/os/win/time_win.c"
    "${OBSERVER_ROOT}/src/os/win/proc_self_win.c")
if(NOT TARGET test_toolchain_process_owned)
    include("${CMAKE_CURRENT_LIST_DIR}/../toolchain_process_owned.cmake")
endif()
foreach(t image_observer_dll test_process_image_observer)
    set_target_properties(${t} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_include_directories(${t} PRIVATE "${OBSERVER_ROOT}/src")
    target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${t} PRIVATE /W4 /WX /utf-8)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(${t} PRIVATE /experimental:c11atomics)
    endif()
endforeach()
enable_testing()
add_test(NAME process_image_observer COMMAND test_process_image_observer "$<TARGET_FILE:image_observer_dll>")
set_tests_properties(process_image_observer PROPERTIES TIMEOUT 120)
if(COMMAND xray_stage_windows_asan_runtime)
    xray_stage_windows_asan_runtime("${CMAKE_CURRENT_BINARY_DIR}")
    xray_prepend_windows_asan_runtime_to_directory_tests()
endif()
