cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(NativeAdmissionOwner C)
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Windows Ninja is required")
endif()
if(NOT TARGET xray_xir_native_admission)
    if(NOT IS_ABSOLUTE "${ADMISSION_SOURCE_BUILD}" OR NOT IS_ABSOLUTE "${ADMISSION_SDK}")
        message(FATAL_ERROR "Provide qualified ADMISSION_SOURCE_BUILD and ADMISSION_SDK")
    endif()
    set(admission_libraries)
    foreach(item src/xir/xray_xir_source_product src/xir/xray_xir_source src/xir/xray_xir_vm
        src/xir/xray_xir_scalar src/xir/xray_xir_cgen src/xir/xray_xir src/xir/xray_xir_admission
        src/xir/xray_xir_declarations src/xir/xray_compiler_parser src/base/xray_compile_resources
        src/toolchain/xray_xir_native_projection src/toolchain/xray_xir_native_artifact
        src/toolchain/xray_toolchain_binding src/toolchain/xray_xir_runtime_sdk
        src/base/xray_json_cursor xray_xir_target xray_toolchain_dependencies xray_toolchain_process xray_xir_workspace
        xray_xir_invocation)
        get_filename_component(name "${item}" NAME)
        add_library(${name} STATIC IMPORTED)
        set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${ADMISSION_SOURCE_BUILD}/${item}.lib")
        list(APPEND admission_libraries ${name})
    endforeach()
    add_library(xray_xir_native_admission STATIC "${XR_ROOT}/src/app/toolchain/xtc_xir_native_admission.c"
        "${XR_ROOT}/src/app/toolchain/xtc_xir_native_operation.c" "${XR_ROOT}/src/app/toolchain/xtc_xir_pe_version.c")
    target_link_libraries(xray_xir_native_admission PUBLIC ${admission_libraries} ntdll bcrypt)
    target_include_directories(xray_xir_native_admission PRIVATE "${XR_ROOT}/src")
    target_compile_options(xray_xir_native_admission PRIVATE /W4 /WX /utf-8)
    target_compile_definitions(xray_xir_native_admission PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
enable_testing()
foreach(mode production injected)
    add_executable(test_native_admission_${mode} test_native_admission.c)
    target_link_libraries(test_native_admission_${mode} PRIVATE xray_xir_native_admission ntdll bcrypt)
    target_include_directories(test_native_admission_${mode} PRIVATE "${XR_ROOT}/src" "${XR_ROOT}/include" "${ADMISSION_SDK}")
    target_compile_options(test_native_admission_${mode} PRIVATE /W4 /WX /utf-8)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(test_native_admission_${mode} PRIVATE /experimental:c11atomics)
    endif()
    target_compile_definitions(test_native_admission_${mode} PRIVATE NDEBUG WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    set_target_properties(test_native_admission_${mode} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(mode STREQUAL "injected")
        target_compile_definitions(test_native_admission_${mode} PRIVATE ADMISSION_INJECTED)
    endif()
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_program(ADMISSION_MSVC NAMES cl REQUIRED)
find_program(ADMISSION_LINKER NAMES link HINTS "${ADMISSION_MSVC}/.." NO_DEFAULT_PATH)
if(NOT ADMISSION_LINKER)
    find_program(ADMISSION_LINKER NAMES link REQUIRED)
endif()
foreach(mode unit native)
    if(mode STREQUAL "unit")
        set(executable test_native_admission_injected)
    else()
        set(executable test_native_admission_production)
    endif()
    add_test(NAME native_admission_${mode} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_native_admission.py" "$<TARGET_FILE:${executable}>"
        "${XR_ROOT}" "${ADMISSION_MSVC}" "${ADMISSION_LINKER}" "${ADMISSION_SDK}" "${mode}")
    set_tests_properties(native_admission_${mode} PROPERTIES TIMEOUT 300 RUN_SERIAL TRUE)
endforeach()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_subdirectory("${XR_ROOT}/tests/unit/cli/pe_version_owner" pe_version)
endif()
