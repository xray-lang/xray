cmake_minimum_required(VERSION 3.25)
project(XirCheckCommandOwner C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "The check command gate requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT IS_ABSOLUTE "${CHECK_SOURCE_BUILD}")
    message(FATAL_ERROR "Set CHECK_SOURCE_BUILD to the fresh production archive directory")
endif()
set(check_libraries "${CHECK_SOURCE_BUILD}/xray_cli_source.lib"
    "${CHECK_SOURCE_BUILD}/src/base/xray_compile_resources.lib")
foreach(name xray_xir_source_product xray_xir_source xray_xir_vm xray_xir_cgen xray_xir xray_compiler_parser xray_xir_scalar xray_xir_admission xray_xir_declarations)
    list(APPEND check_libraries "${CHECK_SOURCE_BUILD}/src/xir/${name}.lib")
endforeach()
foreach(library IN LISTS check_libraries)
    if(NOT EXISTS "${library}")
        message(FATAL_ERROR "Missing production archive: ${library}")
    endif()
endforeach()
set(cli_sources "${XR_ROOT}/src/app/cli/xcli_diag.c" "${XR_ROOT}/src/app/cli/xcli_spec.c")
add_executable(check_owner_driver check_owner_driver.c "${XR_ROOT}/src/app/cli/xcmd_check.c" ${cli_sources})
add_executable(check_owner_injected check_owner_driver.c ${cli_sources})
target_compile_definitions(check_owner_injected PRIVATE CHECK_OWNER_INJECTED=1)
foreach(target check_owner_driver check_owner_injected)
    target_include_directories(${target} PRIVATE "${XR_ROOT}/src" "${XR_ROOT}/include")
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    target_link_libraries(${target} PRIVATE ${check_libraries} bcrypt)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(${target} PRIVATE /experimental:c11atomics)
    endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME check_source_owner COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${CMAKE_CURRENT_LIST_DIR}/check_owner_cases.py" --root "${XR_ROOT}"
    --driver "$<TARGET_FILE:check_owner_driver>" --injected "$<TARGET_FILE:check_owner_injected>"
    --output "${CMAKE_CURRENT_BINARY_DIR}/cases")
set_tests_properties(check_source_owner PROPERTIES TIMEOUT 120 COST 3)
