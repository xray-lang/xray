cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(PeVersionOwner C)
endif()
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This gate requires Ninja")
endif()
get_filename_component(PE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
enable_testing()
add_executable(test_pe_version_production test_pe_version.c)
if(TARGET xray_xir_pe_version)
    target_link_libraries(test_pe_version_production PRIVATE xray_xir_pe_version)
else()
    target_sources(test_pe_version_production PRIVATE "${PE_ROOT}/src/app/toolchain/xtc_xir_pe_version.c")
endif()
target_include_directories(test_pe_version_production PRIVATE "${PE_ROOT}/src")
target_compile_definitions(test_pe_version_production PRIVATE _CRT_SECURE_NO_WARNINGS)
set_target_properties(test_pe_version_production PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(test_pe_version_production PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(test_pe_version_production PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME pe_version_production COMMAND test_pe_version_production)
if(WIN32)
    if(NOT TARGET Python3::Interpreter)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
    endif()
    find_program(PE_VERSION_CL NAMES cl REQUIRED)
    get_filename_component(PE_VERSION_BIN "${PE_VERSION_CL}" DIRECTORY)
    find_program(PE_VERSION_LINK NAMES link HINTS "${PE_VERSION_BIN}" NO_DEFAULT_PATH REQUIRED)
    add_test(NAME pe_version_real_files COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_LIST_DIR}/test_pe_version.py" "$<TARGET_FILE:test_pe_version_production>"
        "${PE_VERSION_CL}" "${PE_VERSION_LINK}" "${CMAKE_CURRENT_BINARY_DIR}/real-fixtures")
endif()
