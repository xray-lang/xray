cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(PeVersionOwner C)
endif()
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This gate requires Ninja")
endif()
get_filename_component(PE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
enable_testing()
foreach(mode production injected)
    add_executable(test_pe_version_${mode} test_pe_version.c)
    if(TARGET xray_xir_pe_version)
        target_link_libraries(test_pe_version_${mode} PRIVATE xray_xir_pe_version)
    else()
        target_sources(test_pe_version_${mode} PRIVATE "${PE_ROOT}/src/app/toolchain/xtc_xir_pe_version.c")
    endif()
    if(mode STREQUAL "production")
        if(TARGET xray_compile_resources)
            target_link_libraries(test_pe_version_${mode} PRIVATE xray_compile_resources)
        else()
            target_sources(test_pe_version_${mode} PRIVATE "${PE_ROOT}/src/base/xcompile_resources.c")
        endif()
        target_compile_definitions(test_pe_version_${mode} PRIVATE PE_VERSION_PRODUCTION)
    endif()
    target_include_directories(test_pe_version_${mode} PRIVATE "${PE_ROOT}/src")
    target_compile_definitions(test_pe_version_${mode} PRIVATE _CRT_SECURE_NO_WARNINGS)
    set_target_properties(test_pe_version_${mode} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(test_pe_version_${mode} PRIVATE /W4 /WX /utf-8)
        if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
            target_compile_options(test_pe_version_${mode} PRIVATE /experimental:c11atomics)
        endif()
    else()
        target_compile_options(test_pe_version_${mode} PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME pe_version_${mode} COMMAND test_pe_version_${mode})
endforeach()
if(WIN32)
    if(NOT TARGET Python3::Interpreter)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
    endif()
    find_program(PE_VERSION_CL NAMES cl REQUIRED)
    get_filename_component(PE_VERSION_BIN "${PE_VERSION_CL}" DIRECTORY)
    find_program(PE_VERSION_LINK NAMES link HINTS "${PE_VERSION_BIN}" NO_DEFAULT_PATH REQUIRED)
    add_test(NAME pe_version_real_files COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_LIST_DIR}/test_pe_version.py" "$<TARGET_FILE:test_pe_version_production>"
        "${PE_VERSION_CL}" "${PE_VERSION_LINK}" "${CMAKE_CURRENT_BINARY_DIR}/real-fixtures")
endif()
