cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(XirFileLeaseOwner C)
endif()
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This gate requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(LEASE_SANITIZE "Enable Clang ASan and UBSan" OFF)
if(LEASE_SANITIZE)
    if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
        message(FATAL_ERROR "Lease sanitizers require clang-cl")
    endif()
    execute_process(COMMAND "${CMAKE_C_COMPILER}" /clang:-print-resource-dir
        OUTPUT_VARIABLE resource OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
endif()
foreach(mode injected production)
    add_executable(test_file_lease_${mode} test_file_lease_owner.c)
    target_include_directories(test_file_lease_${mode} PRIVATE "${XR_ROOT}/src" "${XR_ROOT}/include")
    if(mode STREQUAL "production" AND TARGET xray_xir_target)
        target_link_libraries(test_file_lease_${mode} PRIVATE xray_xir_target)
    else()
        target_sources(test_file_lease_${mode} PRIVATE "${XR_ROOT}/src/base/xutf8.c" "${XR_ROOT}/src/base/xsha256.c")
        if(mode STREQUAL "production")
            target_sources(test_file_lease_${mode} PRIVATE "${XR_ROOT}/src/app/toolchain/xtc_xir_target.c"
                "${XR_ROOT}/src/app/toolchain/xtc_xir_sysroot.c" "${XR_ROOT}/src/app/toolchain/xtc_xir_images.c")
        endif()
    endif()
    if(mode STREQUAL "production")
        target_compile_definitions(test_file_lease_${mode} PRIVATE LEASE_PRODUCTION)
    endif()
    target_compile_definitions(test_file_lease_${mode} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(test_file_lease_${mode} PRIVATE /W4 /WX /utf-8)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(test_file_lease_${mode} PRIVATE /experimental:c11atomics)
    endif()
    if(LEASE_SANITIZE)
        target_compile_options(test_file_lease_${mode} PRIVATE /fsanitize=address /Oy-
            /clang:-fsanitize=undefined /clang:-fno-sanitize=function)
        target_link_options(test_file_lease_${mode} PRIVATE /INCREMENTAL:NO /INCLUDE:__asan_seh_interceptor
            "/WHOLEARCHIVE:${resource}/lib/windows/clang_rt.asan_dynamic_runtime_thunk-x86_64.lib")
        target_link_libraries(test_file_lease_${mode} PRIVATE "${resource}/lib/windows/clang_rt.asan_dynamic-x86_64.lib")
    endif()
endforeach()
if(COMMAND xray_stage_windows_asan_runtime)
    xray_stage_windows_asan_runtime("${CMAKE_CURRENT_BINARY_DIR}")
    cmake_language(DEFER CALL xray_prepend_windows_asan_runtime_to_directory_tests)
endif()
if(NOT TARGET Python3::Interpreter)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
endif()
enable_testing()
foreach(mode injected production)
    add_test(NAME file_lease_${mode} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_file_lease_owner.py" $<TARGET_FILE:test_file_lease_${mode}>)
    set_tests_properties(file_lease_${mode} PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE LABELS "unit;xir;ownership;budget")
    if(LEASE_SANITIZE)
        set_tests_properties(file_lease_${mode} PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:${resource}/lib/windows")
    endif()
endforeach()
