if(TARGET test_source_type_facts)
    return()
endif()
add_executable(test_source_type_facts "${CMAKE_CURRENT_LIST_DIR}/test_types.c")
target_link_libraries(test_source_type_facts PRIVATE xray_lsp_source_workspace)
target_include_directories(test_source_type_facts PRIVATE "${PROJECT_SOURCE_DIR}/tests/unit/xir")
target_compile_features(test_source_type_facts PRIVATE c_std_11)
add_executable(test_source_type_facts_owner "${CMAKE_CURRENT_LIST_DIR}/test_types_owner.c")
target_link_libraries(test_source_type_facts_owner PRIVATE xray_xir_source)
target_include_directories(test_source_type_facts_owner PRIVATE "${PROJECT_SOURCE_DIR}/tests/unit/xir")
foreach(target test_source_type_facts test_source_type_facts_owner)
    if(WIN32)
        target_link_libraries(${target} PRIVATE bcrypt)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME test_source_type_facts COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${CMAKE_CURRENT_LIST_DIR}/test_types.py" "$<TARGET_FILE:test_source_type_facts>" "${PROJECT_SOURCE_DIR}/stdlib")
add_test(NAME test_source_type_inheritance_required COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${CMAKE_CURRENT_LIST_DIR}/test_types.py" "$<TARGET_FILE:test_source_type_facts>" "${PROJECT_SOURCE_DIR}/stdlib" --inheritance-required)
set_tests_properties(test_source_type_facts test_source_type_inheritance_required PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE LABELS "unit;source;ownership;metadata;required")
add_test(NAME test_source_type_facts_owner COMMAND test_source_type_facts_owner)
set_tests_properties(test_source_type_facts_owner PROPERTIES TIMEOUT 30 LABELS "unit;source;ownership;allocation")
add_test(NAME test_source_type_facts_resource_probe COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${CMAKE_CURRENT_LIST_DIR}/test_resources.py" "$<TARGET_FILE:test_source_type_facts>" "${PROJECT_SOURCE_DIR}/stdlib")
set_tests_properties(test_source_type_facts_resource_probe PROPERTIES TIMEOUT 300 RUN_SERIAL TRUE LABELS "unit;source;ownership;allocation")
add_test(NAME test_source_type_facts_resources COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${CMAKE_CURRENT_LIST_DIR}/test_resources.py" "$<TARGET_FILE:test_source_type_facts>" "${PROJECT_SOURCE_DIR}/stdlib" --full)
