function(json_clone_fixture target)
    add_executable(${target} "${CMAKE_CURRENT_LIST_DIR}/test_clone.c"
        "${CMAKE_CURRENT_LIST_DIR}/test_allocator.c" "${CMAKE_CURRENT_LIST_DIR}/json_owner.c"
        "${PROJECT_SOURCE_DIR}/src/base/xutf8.c")
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/src" "${PROJECT_SOURCE_DIR}/src/base")
    target_compile_features(${target} PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
endfunction()
json_clone_fixture(test_lsp_json_clone)
json_clone_fixture(test_lsp_json_clone_before)
target_compile_definitions(test_lsp_json_clone_before PRIVATE CLONE_BEFORE=1)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(mode normal full before)
    if(mode STREQUAL "full")
        set(name test_lsp_json_clone_resources)
        set(binary test_lsp_json_clone)
    elseif(mode STREQUAL "before")
        set(name test_lsp_json_clone_counterfactual)
        set(binary test_lsp_json_clone_before)
    else()
        set(name test_lsp_json_clone_normal)
        set(binary test_lsp_json_clone)
    endif()
    add_test(NAME ${name} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_clone.py" "$<TARGET_FILE:${binary}>" ${mode})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE LABELS "unit;lsp;json;ownership;allocation")
endforeach()
