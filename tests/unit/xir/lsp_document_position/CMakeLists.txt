if(TARGET test_lsp_document_position)
    return()
endif()
add_executable(test_lsp_document_position "${CMAKE_CURRENT_LIST_DIR}/test_position.c")
add_executable(test_lsp_document_position_baseline "${CMAKE_CURRENT_LIST_DIR}/test_position.c"
    "${CMAKE_CURRENT_LIST_DIR}/baseline.c")
foreach(position_target test_lsp_document_position test_lsp_document_position_baseline)
    target_link_libraries(${position_target} PRIVATE xray_lsp_document_open)
    target_include_directories(${position_target} PRIVATE "${PROJECT_SOURCE_DIR}/tests/unit/xir")
    target_compile_features(${position_target} PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(${position_target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${position_target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
if(MSVC)
    target_compile_options(test_lsp_document_position PRIVATE /WX)
endif()
add_test(NAME test_lsp_document_position COMMAND test_lsp_document_position)
set_tests_properties(test_lsp_document_position PROPERTIES TIMEOUT 30 LABELS "unit;lsp;utf16;ownership")
