    return()

find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(target test_lsp_parse_diagnostics test_lsp_parse_diagnostics_before)
    add_executable(${target} "${CMAKE_CURRENT_LIST_DIR}/test_diagnostics.c")
    target_link_libraries(${target} PRIVATE xray_lsp_document_open)
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/tests/unit/xir" "${PROJECT_SOURCE_DIR}/src/module")
    target_compile_features(${target} PRIVATE c_std_11)
    if(WIN32)
        target_link_libraries(${target} PRIVATE bcrypt)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
target_sources(test_lsp_parse_diagnostics_before PRIVATE "${CMAKE_CURRENT_LIST_DIR}/before_graph.c")
target_compile_definitions(test_lsp_parse_diagnostics_before PRIVATE XR_PARSE_DIAGNOSTIC_BEFORE=1)
foreach(mode normal resources before)
    if(mode STREQUAL "before")
        set(image test_lsp_parse_diagnostics_before)
    else()
        set(image test_lsp_parse_diagnostics)
    endif()
    add_test(NAME test_lsp_parse_diagnostics_${mode} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_diagnostics.py" "$<TARGET_FILE:${image}>" "${PROJECT_SOURCE_DIR}/stdlib" --${mode})
    set_tests_properties(test_lsp_parse_diagnostics_${mode} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE LABELS "unit;lsp;source;ownership;allocation")
endforeach()
