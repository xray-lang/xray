cmake_minimum_required(VERSION 3.20)
project(P19UcrtNoargProvider LANGUAGES C)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The reviewed provider is MSVC x64 UCRT")
endif()
add_executable(p19_ucrt_noarg_probe probe.c)
target_compile_features(p19_ucrt_noarg_probe PRIVATE c_std_11)
target_compile_options(p19_ucrt_noarg_probe PRIVATE /std:c11 /W4 /WX)
set_property(TARGET p19_ucrt_noarg_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
