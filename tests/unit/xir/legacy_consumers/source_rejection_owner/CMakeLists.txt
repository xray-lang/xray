cmake_minimum_required(VERSION 3.25)
project(LegacySourceRejectionOwner C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Ninja is the only supported generator")
endif()
if(NOT XRAY_SOURCE_ROOT OR NOT EXISTS "${XRAY_PRODUCER_ARCHIVES}")
    message(FATAL_ERROR "An explicit source root and producer archive manifest are required")
endif()
include("${XRAY_PRODUCER_ARCHIVES}")
if(NOT XR_PRODUCER_ARCHIVES)
    message(FATAL_ERROR "The producer archive manifest is empty")
endif()
add_executable(legacy_source_rejection_owner source_rejection_owner.c)
add_executable(legacy_source_rejection_faults source_rejection_faults.c)
foreach(target IN ITEMS legacy_source_rejection_owner legacy_source_rejection_faults)
target_link_libraries(${target} PRIVATE ${XR_PRODUCER_ARCHIVES})
target_include_directories(${target} PRIVATE "${XRAY_SOURCE_ROOT}/src" "${XRAY_SOURCE_ROOT}/include")
set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /experimental:c11atomics)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
endif()

endforeach()
target_include_directories(legacy_source_rejection_faults PRIVATE "${XRAY_SOURCE_ROOT}/tests/unit/xir")
