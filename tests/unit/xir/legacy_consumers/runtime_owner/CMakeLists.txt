cmake_minimum_required(VERSION 3.25)
project(LegacyRuntimeOwner C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This actual provider lane requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
if(NOT XR_PRODUCT_BUILD)
    message(FATAL_ERROR "Supply the fresh same-source production build")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${CMAKE_CURRENT_LIST_DIR}/producer_archives.py" --build "${XR_PRODUCT_BUILD}"
    --output "${CMAKE_CURRENT_BINARY_DIR}/producer-archives.cmake"
    COMMAND_ERROR_IS_FATAL ANY)
include("${CMAKE_CURRENT_BINARY_DIR}/producer-archives.cmake")
add_executable(legacy_runtime_owner test_runtime_owner.c)
target_link_libraries(legacy_runtime_owner PRIVATE ${XR_PRODUCER_ARCHIVES})
target_include_directories(legacy_runtime_owner PRIVATE "${XR_ROOT}/include" "${XR_ROOT}/src" "${XR_ROOT}/tests/unit")
target_compile_definitions(legacy_runtime_owner PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(legacy_runtime_owner PRIVATE /W4 /WX /utf-8)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(legacy_runtime_owner PRIVATE /experimental:c11atomics)
endif()
if(XR_ASAN)
    target_compile_options(legacy_runtime_owner PRIVATE -fsanitize=address,undefined)
    target_link_options(legacy_runtime_owner PRIVATE -fsanitize=address,undefined)
endif()
