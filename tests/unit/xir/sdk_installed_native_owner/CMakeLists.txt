cmake_minimum_required(VERSION 3.25)
project(InstalledNativeSdkOwner C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This installed SDK consumer requires Windows and Ninja")
endif()
if(NOT IS_ABSOLUTE "${XIR_SDK_BUNDLE_DIR}" OR
   NOT EXISTS "${XIR_SDK_BUNDLE_DIR}/sdk_manifest.json")
    message(FATAL_ERROR "Provide the actual relocated runtime SDK")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(installed_native_owner consumer.c native-program.c)
set_target_properties(installed_native_owner PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_compile_definitions(installed_native_owner PRIVATE
    NDEBUG NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(installed_native_owner PRIVATE /W4 /WX /utf-8)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(installed_native_owner PRIVATE /experimental:c11atomics)
endif()
target_include_directories(installed_native_owner PRIVATE
    "${XIR_SDK_BUNDLE_DIR}/src" "${XIR_SDK_BUNDLE_DIR}/include"
    "${XIR_SDK_BUNDLE_DIR}/generated")
foreach(owner xray_compile_resources xray_xir_admission xray_xir_declarations
              xray_xir_scalar xray_xir_runtime_host)
    set(path "${XIR_SDK_BUNDLE_DIR}/lib/${owner}.lib")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Missing installed production archive: ${path}")
    endif()
    add_library(sdk_${owner} STATIC IMPORTED)
    set_target_properties(sdk_${owner} PROPERTIES IMPORTED_LOCATION "${path}")
    target_link_libraries(installed_native_owner PRIVATE sdk_${owner})
endforeach()
target_link_libraries(installed_native_owner PRIVATE kernel32)
