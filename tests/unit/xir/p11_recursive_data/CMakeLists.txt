# Full recursive data and original-owner consumers use ordinary targets.
function(xray_recursive_consumer target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE xray_xir_source xray_xir_vm)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/..")
    target_compile_definitions(${target} PRIVATE
        MATERIALIZATION_ROOT="${CMAKE_CURRENT_LIST_DIR}/../p11_materialization"
        DEEP_ROOT="${CMAKE_CURRENT_LIST_DIR}/fixtures"
        CONTROL_ROOT="${CMAKE_CURRENT_LIST_DIR}/fixtures")
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endfunction()

function(xray_recursive_test name target seconds)
    add_test(NAME ${name} COMMAND ${target} ${ARGN})
    set_tests_properties(${name} PROPERTIES TIMEOUT ${seconds} LABELS "unit;xir;ctfe;ownership;recursive-data")
    if(seconds EQUAL 600)
        set_tests_properties(${name} PROPERTIES RUN_SERIAL TRUE LABELS "unit;xir;ctfe;ownership;recursive-data;compiler-fault;resources")
    endif()
endfunction()
