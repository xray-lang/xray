cmake_minimum_required(VERSION 3.24)
project(lsp_protocol_message_qualification LANGUAGES C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Ninja is required")
endif()
get_filename_component(XR_PROTOCOL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(test_lsp_protocol_message test_protocol_message.c "${XR_PROTOCOL_ROOT}/src/base/xutf8.c")
target_include_directories(test_lsp_protocol_message PRIVATE "${XR_PROTOCOL_ROOT}/src" "${XR_PROTOCOL_ROOT}/src/base")
target_compile_features(test_lsp_protocol_message PRIVATE c_std_11)
target_compile_definitions(test_lsp_protocol_message PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(test_lsp_protocol_message PRIVATE /W4 /WX /utf-8)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(test_lsp_protocol_message PRIVATE /experimental:c11atomics)
    endif()
else()
    target_compile_options(test_lsp_protocol_message PRIVATE -Wall -Wextra -Werror)
endif()
option(ENABLE_ASAN "Address sanitizer" OFF)
option(ENABLE_UBSAN "Undefined behavior sanitizer" OFF)
if(ENABLE_ASAN)
    if(MSVC)
        string(REPLACE "/RTC1" "" CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG}")
        set_property(TARGET test_lsp_protocol_message PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
        target_compile_options(test_lsp_protocol_message PRIVATE /fsanitize=address /Oy-)
        target_link_options(test_lsp_protocol_message PRIVATE /INCREMENTAL:NO)
        if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
            set(XR_PROTOCOL_SANITIZER_RUNTIME_DIR "" CACHE PATH "Clang Windows sanitizer runtime directory")
            foreach(runtime clang_rt.asan_dynamic-x86_64.lib clang_rt.asan_dynamic_runtime_thunk-x86_64.lib)
                if(NOT EXISTS "${XR_PROTOCOL_SANITIZER_RUNTIME_DIR}/${runtime}")
                    message(FATAL_ERROR "Missing sanitizer runtime ${runtime}")
                endif()
            endforeach()
            target_link_options(test_lsp_protocol_message PRIVATE
                "${XR_PROTOCOL_SANITIZER_RUNTIME_DIR}/clang_rt.asan_dynamic-x86_64.lib"
                /INCLUDE:__asan_seh_interceptor
                "/WHOLEARCHIVE:${XR_PROTOCOL_SANITIZER_RUNTIME_DIR}/clang_rt.asan_dynamic_runtime_thunk-x86_64.lib")
        endif()
    else()
        target_compile_options(test_lsp_protocol_message PRIVATE -fsanitize=address -fno-omit-frame-pointer)
        target_link_options(test_lsp_protocol_message PRIVATE -fsanitize=address)
    endif()
endif()
if(ENABLE_UBSAN)
    if(MSVC)
        if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang" OR NOT ENABLE_ASAN)
            message(FATAL_ERROR "Combined clang-cl ASan/UBSan is required for this Windows profile")
        endif()
        target_compile_options(test_lsp_protocol_message PRIVATE /clang:-fsanitize=undefined /clang:-fno-sanitize=function /Oy-)
    else()
        target_compile_options(test_lsp_protocol_message PRIVATE -fsanitize=undefined -fno-sanitize=function)
        target_link_options(test_lsp_protocol_message PRIVATE -fsanitize=undefined -fno-sanitize=function)
    endif()
endif()
enable_testing()
add_test(NAME test_lsp_protocol_message COMMAND test_lsp_protocol_message)
set_tests_properties(test_lsp_protocol_message PROPERTIES TIMEOUT 30 LABELS "unit;lsp;ownership;allocation")
