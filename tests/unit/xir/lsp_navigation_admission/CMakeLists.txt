if(TARGET test_lsp_navigation_admission)
    return()
endif()
source_task_target(test_lsp_navigation_admission "${PROJECT_SOURCE_DIR}/src/app/lsp/xlsp_navigation.c")
source_task_target(test_lsp_navigation_admission_before "${CMAKE_CURRENT_LIST_DIR}/before_navigation.c")
target_compile_definitions(test_lsp_navigation_admission PRIVATE XLSP_NAVIGATION_ADMISSION=1)
target_compile_definitions(test_lsp_navigation_admission_before PRIVATE XLSP_NAVIGATION_ADMISSION=1)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(mode normal resources counterfactual)
    if(mode STREQUAL "counterfactual")
        set(image test_lsp_navigation_admission_before)
    else()
        set(image test_lsp_navigation_admission)
    endif()
    add_test(NAME test_lsp_navigation_admission_${mode} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_admission.py" "$<TARGET_FILE:${image}>" "${PROJECT_SOURCE_DIR}/stdlib" --${mode})
    set_tests_properties(test_lsp_navigation_admission_${mode} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE LABELS "unit;lsp;source;ownership;protocol")
endforeach()
foreach(mode normal resources)
    add_test(NAME test_lsp_navigation_worker_${mode} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_worker.py" "$<TARGET_FILE:test_lsp_navigation_admission>" "${PROJECT_SOURCE_DIR}/stdlib" --${mode})
    set_tests_properties(test_lsp_navigation_worker_${mode} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE LABELS "unit;lsp;source;ownership;allocation")
endforeach()
set_tests_properties(test_lsp_navigation_worker_resources PROPERTIES TIMEOUT 300 PROCESSORS 8 COST 290)
