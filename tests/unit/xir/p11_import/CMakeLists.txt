add_executable(test_xir_p11_import test_import.c)
target_link_libraries(test_xir_p11_import PRIVATE xray_xir_source xray_xir_vm)
target_include_directories(test_xir_p11_import PRIVATE "${CMAKE_CURRENT_LIST_DIR}/..")
target_compile_definitions(test_xir_p11_import PRIVATE MATERIALIZATION_ROOT="${CMAKE_CURRENT_LIST_DIR}/../p11_materialization")
set_target_properties(test_xir_p11_import PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
    target_compile_options(test_xir_p11_import PRIVATE /utf-8 /W4 /WX)
else()
    target_compile_options(test_xir_p11_import PRIVATE -Wall -Wextra -Werror)
endif()
