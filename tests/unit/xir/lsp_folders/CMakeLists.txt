if(TARGET test_lsp_source_folders)
    return()
endif()
add_executable(test_lsp_source_folders "${CMAKE_CURRENT_LIST_DIR}/test_folders.c"
    "${PROJECT_SOURCE_DIR}/src/app/lsp/xlsp_index_pool.c"
    "${PROJECT_SOURCE_DIR}/src/app/lsp/xlsp_source_index.c"
    "${PROJECT_SOURCE_DIR}/src/app/lsp/xlsp_source_index_store.c")
target_link_libraries(test_lsp_source_folders PRIVATE xray_lsp_document_open)
if(WIN32)
    target_sources(test_lsp_source_folders PRIVATE "${PROJECT_SOURCE_DIR}/src/os/win/thread_win.c" "${PROJECT_SOURCE_DIR}/src/os/win/time_win.c")
    target_link_libraries(test_lsp_source_folders PRIVATE bcrypt)
else()
    target_sources(test_lsp_source_folders PRIVATE "${PROJECT_SOURCE_DIR}/src/os/unix/thread_unix.c" "${PROJECT_SOURCE_DIR}/src/os/unix/time_unix.c")
endif()
target_include_directories(test_lsp_source_folders PRIVATE "${PROJECT_SOURCE_DIR}/tests/unit/xir" "${PROJECT_SOURCE_DIR}/src/base")
target_compile_features(test_lsp_source_folders PRIVATE c_std_11)
if(MSVC)
    target_compile_options(test_lsp_source_folders PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(test_lsp_source_folders PRIVATE -Wall -Wextra -Werror)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME test_lsp_source_folders COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${CMAKE_CURRENT_LIST_DIR}/test_folders.py" "$<TARGET_FILE:test_lsp_source_folders>" "${PROJECT_SOURCE_DIR}/stdlib")
set_tests_properties(test_lsp_source_folders PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE LABELS "unit;lsp;ownership;allocation")
