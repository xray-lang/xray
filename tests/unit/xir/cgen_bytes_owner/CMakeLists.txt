cmake_minimum_required(VERSION 3.25)
project(XirCgenBytesOwner C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This gate requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT IS_ABSOLUTE "${CGEN_SOURCE_BUILD}")
    message(FATAL_ERROR "Set CGEN_SOURCE_BUILD to the fresh formal build")
endif()
add_executable(test_cgen_bytes_owner test_cgen_bytes_owner.c)
target_include_directories(test_cgen_bytes_owner PRIVATE "${XR_ROOT}/src" "${XR_ROOT}/include")
target_compile_definitions(test_cgen_bytes_owner PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(test_cgen_bytes_owner PRIVATE /W4 /WX /utf-8)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(test_cgen_bytes_owner PRIVATE /experimental:c11atomics)
endif()
foreach(name xray_xir_cgen xray_xir_scalar xray_xir xray_xir_admission xray_xir_declarations)
    target_link_libraries(test_cgen_bytes_owner PRIVATE "${CGEN_SOURCE_BUILD}/src/xir/${name}.lib")
endforeach()
enable_testing()
add_test(NAME cgen_bytes_owner COMMAND test_cgen_bytes_owner)
set_tests_properties(cgen_bytes_owner PROPERTIES TIMEOUT 30)
