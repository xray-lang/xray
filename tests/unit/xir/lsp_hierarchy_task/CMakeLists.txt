if(TARGET test_lsp_source_params)
    return()
endif()
add_executable(test_lsp_source_params "${CMAKE_CURRENT_LIST_DIR}/test_params.c")
target_link_libraries(test_lsp_source_params PRIVATE xray_lsp_hierarchy_handlers)
target_include_directories(test_lsp_source_params PRIVATE "${PROJECT_SOURCE_DIR}/tests/unit/xir")
target_compile_features(test_lsp_source_params PRIVATE c_std_11)
if(MSVC)
    target_compile_options(test_lsp_source_params PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(test_lsp_source_params PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME test_lsp_source_params_normal COMMAND test_lsp_source_params)
add_test(NAME test_lsp_source_params_resources COMMAND test_lsp_source_params --full)
set_tests_properties(test_lsp_source_params_normal test_lsp_source_params_resources
    PROPERTIES TIMEOUT 30 LABELS "unit;lsp;source;ownership;allocation")
source_task_target(test_lsp_hierarchy_task "${PROJECT_SOURCE_DIR}/src/app/lsp/xlsp_navigation.c")
source_task_target(test_lsp_hierarchy_task_before "${PROJECT_SOURCE_DIR}/src/app/lsp/xlsp_navigation.c"
    "${CMAKE_CURRENT_LIST_DIR}/before_task_owner.c")
target_compile_definitions(test_lsp_hierarchy_task PRIVATE XLSP_HIERARCHY_TASK=1)
target_compile_definitions(test_lsp_hierarchy_task_before PRIVATE XLSP_HIERARCHY_TASK=1)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(mode normal capture counterfactual)
    if(mode STREQUAL "counterfactual")
        set(image test_lsp_hierarchy_task_before)
    else()
        set(image test_lsp_hierarchy_task)
    endif()
    add_test(NAME test_lsp_hierarchy_task_${mode} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_task.py" "$<TARGET_FILE:${image}>" "${PROJECT_SOURCE_DIR}/stdlib" --${mode})
    set_tests_properties(test_lsp_hierarchy_task_${mode} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE
        LABELS "unit;lsp;source;ownership")
endforeach()
