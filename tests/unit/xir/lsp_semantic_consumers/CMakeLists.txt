if(TARGET test_lsp_semantic_provider)
    return()
endif()
add_executable(test_lsp_semantic_provider "${CMAKE_CURRENT_LIST_DIR}/test_provider.c"
    "${PROJECT_SOURCE_DIR}/src/app/lsp/xlsp_semantic_tokens.c" "${PROJECT_SOURCE_DIR}/src/base/xutf8.c")
add_executable(test_lsp_semantic_source "${CMAKE_CURRENT_LIST_DIR}/test_source_semantic.c")
target_link_libraries(test_lsp_semantic_source PRIVATE xray_lsp_source_workspace)
if(WIN32)
    target_link_libraries(test_lsp_semantic_source PRIVATE bcrypt)
endif()
foreach(target test_lsp_semantic_provider test_lsp_semantic_source)
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/src" "${PROJECT_SOURCE_DIR}/src/base" "${PROJECT_SOURCE_DIR}/tests/unit/xir")
    target_compile_features(${target} PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME test_lsp_semantic_provider COMMAND test_lsp_semantic_provider)
set_tests_properties(test_lsp_semantic_provider PROPERTIES TIMEOUT 30 LABELS "unit;lsp;ownership;allocation")
set(semantic_cases param_modes quoted_literals enum_fields)
set(semantic_mode 0)
foreach(case IN LISTS semantic_cases)
    add_test(NAME test_lsp_semantic_${case} COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/test_source_semantic.py" "$<TARGET_FILE:test_lsp_semantic_source>" "${PROJECT_SOURCE_DIR}/stdlib" "${semantic_mode}")
    set_tests_properties(test_lsp_semantic_${case} PROPERTIES TIMEOUT 30 LABELS "unit;lsp;ownership;metadata")
    math(EXPR semantic_mode "${semantic_mode}+1")
endforeach()
