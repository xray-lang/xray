cmake_minimum_required(VERSION 3.25)
project(XirHostCliOwner C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "The host CLI gate requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${XR_ROOT}/scripts/gen_xir_runtime_sdk_recipe.py" --root "${XR_ROOT}"
    --output "${CMAKE_CURRENT_BINARY_DIR}/sdk_recipe.inc.c"
    --record "${CMAKE_CURRENT_BINARY_DIR}/runtime_recipe.json"
    COMMAND_ERROR_IS_FATAL ANY)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/runtime_recipe.json" recipe)
string(JSON count LENGTH "${recipe}" archive_source_groups)
math(EXPR last "${count}-1")
set(runtime_sources)
foreach(index RANGE 0 ${last})
    string(JSON owner MEMBER "${recipe}" archive_source_groups ${index})
    string(JSON source_count LENGTH "${recipe}" archive_source_groups "${owner}")
    math(EXPR source_last "${source_count}-1")
    foreach(source_index RANGE 0 ${source_last})
        string(JSON source GET "${recipe}" archive_source_groups "${owner}" ${source_index})
        list(APPEND runtime_sources "${XR_ROOT}/${source}")
    endforeach()
endforeach()
add_library(host_cli_production STATIC ${runtime_sources})
add_executable(test_host_cli_owner test_host_cli_owner.c)
target_link_libraries(test_host_cli_owner PRIVATE host_cli_production)
add_executable(test_value_format_core "${XR_ROOT}/tests/unit/base/test_value_format_core.c")
set(owner_targets host_cli_production test_host_cli_owner test_value_format_core)
if(HOST_SDK_BUNDLE)
    add_executable(host_cli_sdk_lease "${XR_ROOT}/tests/unit/xir/test_xir_sdk_lease.c"
        "${XR_ROOT}/src/toolchain/xr_xir_runtime_sdk.c" "${XR_ROOT}/src/base/xjson_cursor.c")
    target_link_libraries(host_cli_sdk_lease PRIVATE host_cli_production bcrypt)
    target_include_directories(host_cli_sdk_lease PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${HOST_SDK_BUNDLE}")
    list(APPEND owner_targets host_cli_sdk_lease)
endif()
if(HOST_SOURCE_BUILD)
    add_executable(emit_host_cli emit_host_cli.c)
    target_link_libraries(emit_host_cli PRIVATE host_cli_production)
    foreach(name xray_xir_source_product xray_xir_source xray_xir_vm xray_xir_cgen xray_xir xray_compiler_parser)
        set(path "${HOST_SOURCE_BUILD}/src/xir/${name}.lib")
        if(NOT EXISTS "${path}")
            message(FATAL_ERROR "Missing actual Source archive: ${path}")
        endif()
        target_link_libraries(emit_host_cli PRIVATE "${path}")
    endforeach()
    list(APPEND owner_targets emit_host_cli)
endif()
if(HOST_GENERATED_CASE)
    add_library(host_program_cases OBJECT program_cases.c)
    add_executable(test_host_cli_program "$<TARGET_OBJECTS:host_program_cases>" "${HOST_GENERATED_CASE}")
    target_link_libraries(test_host_cli_program PRIVATE host_cli_production)
    list(APPEND owner_targets host_program_cases test_host_cli_program)
    if(HOST_GENERATED_INITIALIZATION)
        add_executable(test_host_cli_initialization "$<TARGET_OBJECTS:host_program_cases>" "${HOST_GENERATED_INITIALIZATION}")
        target_link_libraries(test_host_cli_initialization PRIVATE host_cli_production)
        list(APPEND owner_targets test_host_cli_initialization)
    endif()
endif()
foreach(target ${owner_targets})
    target_include_directories(${target} PRIVATE "${XR_ROOT}/src" "${XR_ROOT}/include")
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
        if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
            target_compile_options(${target} PRIVATE /experimental:c11atomics)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
enable_testing()
add_test(NAME host_cli_owner COMMAND test_host_cli_owner)
add_test(NAME value_format_core COMMAND test_value_format_core)
set_tests_properties(host_cli_owner value_format_core PROPERTIES TIMEOUT 60)
if(TARGET emit_host_cli)
    add_test(NAME host_cli_source_vm COMMAND "${Python3_EXECUTABLE}" -X utf8
        "${CMAKE_CURRENT_LIST_DIR}/source_cases.py" --root "${XR_ROOT}"
        --emitter "$<TARGET_FILE:emit_host_cli>" --output "${CMAKE_CURRENT_BINARY_DIR}/source-cases")
    set_tests_properties(host_cli_source_vm PROPERTIES TIMEOUT 90)
endif()
if(TARGET test_host_cli_program)
    add_test(NAME host_cli_program_oom COMMAND test_host_cli_program oom)
    add_test(NAME host_cli_program_io COMMAND test_host_cli_program io)
    add_test(NAME host_cli_program_owner COMMAND test_host_cli_program owner)
    set_tests_properties(host_cli_program_oom host_cli_program_io host_cli_program_owner PROPERTIES TIMEOUT 120)
endif()
if(TARGET test_host_cli_initialization)
    add_test(NAME host_cli_initialization COMMAND test_host_cli_initialization init)
    set_tests_properties(host_cli_initialization PROPERTIES TIMEOUT 60)
endif()
