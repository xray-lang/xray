cmake_minimum_required(VERSION 3.21)
project(xir_builder_baseline LANGUAGES C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "The baseline requires Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(test_xir_builder_baseline test_xir_builder_baseline.c)
target_include_directories(test_xir_builder_baseline PRIVATE "${XR_ROOT}/src" "${XR_ROOT}/include")
set_target_properties(test_xir_builder_baseline PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
    target_compile_options(test_xir_builder_baseline PRIVATE /utf-8 /W4 /WX)
    target_compile_definitions(test_xir_builder_baseline PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(test_xir_builder_baseline PRIVATE /experimental:c11atomics)
    endif()
else()
    target_compile_options(test_xir_builder_baseline PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
foreach(scenario short digits chunks reserve join snapshot)
    add_test(NAME builder_baseline_${scenario} COMMAND test_xir_builder_baseline ${scenario})
    set_tests_properties(builder_baseline_${scenario} PROPERTIES TIMEOUT 120 LABELS "measurement;legacy-builder")
endforeach()
