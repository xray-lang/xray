cmake_minimum_required(VERSION 3.25)
project(Program29AbiBoundary C)
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Windows Ninja required")
endif()
get_filename_component(ABI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_definitions(NDEBUG NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
add_compile_options(/W4 /WX /utf-8)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    add_compile_options(/experimental:c11atomics)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/old28_headers.zip" archive_hash)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/old28_generated.c" code_hash)
file(READ "${CMAKE_CURRENT_LIST_DIR}/old28_provenance.json" record)
string(JSON wanted_archive GET "${record}" headers sha256)
string(JSON wanted_code GET "${record}" source sha256)
if(NOT archive_hash STREQUAL wanted_archive OR NOT code_hash STREQUAL wanted_code)
    message(FATAL_ERROR "Authentic old producer inputs changed")
endif()
set(old_root "${CMAKE_CURRENT_BINARY_DIR}/old28-headers")
file(MAKE_DIRECTORY "${old_root}")
execute_process(COMMAND "${CMAKE_COMMAND}" -E tar xf "${CMAKE_CURRENT_LIST_DIR}/old28_headers.zip"
    WORKING_DIRECTORY "${old_root}" COMMAND_ERROR_IS_FATAL ANY)
add_library(old28_object OBJECT old28_generated.c old28_guard.c)
target_include_directories(old28_object BEFORE PRIVATE "${old_root}/src" "${old_root}/include")
add_executable(program29_generator producer.c "${ABI_ROOT}/src/xir/xxir_specialize.c"
    "${ABI_ROOT}/src/xir/xxir_emit_c.c" "${ABI_ROOT}/src/aot/xi_cgen_verify_output.c")
set(current_code "${CMAKE_CURRENT_BINARY_DIR}/current29.c")
add_custom_command(OUTPUT "${current_code}" COMMAND program29_generator "${current_code}" current29
    DEPENDS program29_generator VERBATIM)
add_executable(test_program29_boundary test_program29_boundary.c "${current_code}" $<TARGET_OBJECTS:old28_object>)
foreach(owner xray_compile_resources xray_xir_admission xray_xir_declarations xray_xir_scalar xray_xir_runtime_host)
    foreach(target program29_generator test_program29_boundary)
        target_link_libraries(${target} PRIVATE "${XIR_SDK_BUNDLE_DIR}/lib/${owner}.lib")
    endforeach()
endforeach()
foreach(target program29_generator test_program29_boundary)
    target_include_directories(${target} PRIVATE "${ABI_ROOT}/src" "${ABI_ROOT}/include")
    target_link_libraries(${target} PRIVATE kernel32)
endforeach()
if(ENABLE_ASAN)
    if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
        message(FATAL_ERROR "Actual clang-cl sanitizer required")
    endif()
    set(rt "C:/Program Files/LLVM/lib/clang/22/lib/windows")
    foreach(target program29_generator test_program29_boundary old28_object)
        target_compile_definitions(${target} PRIVATE XR_BUILD_ASAN=1 XR_BUILD_UBSAN=1)
        target_compile_options(${target} PRIVATE /fsanitize=address /Oy- /clang:-fsanitize=undefined /clang:-fno-sanitize=function)
    endforeach()
    foreach(target program29_generator test_program29_boundary)
        target_link_options(${target} PRIVATE /INCREMENTAL:NO "${rt}/clang_rt.asan_dynamic-x86_64.lib"
            /INCLUDE:__asan_seh_interceptor "/WHOLEARCHIVE:${rt}/clang_rt.asan_dynamic_runtime_thunk-x86_64.lib")
    endforeach()
endif()
enable_testing()
add_test(NAME test_program29_boundary COMMAND test_program29_boundary)
set_tests_properties(test_program29_boundary PROPERTIES LABELS "unit;xir;abi;ownership;sdk" RUN_SERIAL TRUE TIMEOUT 90)
