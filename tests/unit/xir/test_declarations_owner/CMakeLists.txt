cmake_minimum_required(VERSION 3.21)
project(xray_test_declarations_owner C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Ninja is required")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(PROJECT_SOURCE_DIR "${XR_ROOT}")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(XRAY_PYTHON "${Python3_EXECUTABLE}")
set(XRAY_COMMON_INCLUDES "${XR_ROOT}/src" "${XR_ROOT}/include")
include_directories(${XRAY_COMMON_INCLUDES})
if(MSVC)
    add_compile_options(/W4 /WX /utf-8)
    add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        add_compile_options(/experimental:c11atomics)
    endif()
else()
    add_compile_options(-Wall -Wextra -Werror -pedantic)
endif()
add_custom_target(gen-native-type-defs DEPENDS "${XR_ROOT}/src/shared/xnative_declarations.inc.c")
file(GLOB scripts "${XR_ROOT}/stdlib/*/*.xr")
foreach(path IN LISTS scripts)
    get_filename_component(parent "${path}" DIRECTORY)
    get_filename_component(parent "${parent}" NAME)
    get_filename_component(name "${path}" NAME_WE)
    if(NOT name STREQUAL parent OR parent STREQUAL "types")
        list(REMOVE_ITEM scripts "${path}")
    endif()
endforeach()
file(GLOB_RECURSE generator_inputs "${XR_ROOT}/tools/stdlibgen/*.py" "${XR_ROOT}/stdlib/defs/*.def")
set(embedded "${CMAKE_BINARY_DIR}/generated/stdlib_embedded_sources.inc")
add_custom_command(OUTPUT "${embedded}"
    COMMAND "${Python3_EXECUTABLE}" -X utf8 "${XR_ROOT}/scripts/generate_stdlib_embedded.py"
        --root "${XR_ROOT}" --source-output "${embedded}" ${scripts}
    DEPENDS "${XR_ROOT}/scripts/generate_stdlib_embedded.py" ${generator_inputs}
        "${XR_ROOT}/stdlib/stdlib_boundary.toml" ${scripts} VERBATIM)
add_custom_target(gen-stdlib-embedded-sources DEPENDS "${embedded}")
add_subdirectory("${XR_ROOT}/src/base" base)
add_subdirectory("${XR_ROOT}/src/xir" xir)
enable_testing()
add_executable(test_test_api ../test_xir_compile_api.c)
add_test(NAME test_test_api COMMAND test_test_api)
add_test(NAME meta_ownership_inventory COMMAND "${Python3_EXECUTABLE}" -X utf8
    "${XR_ROOT}/scripts/check_meta_ownership.py" --root "${XR_ROOT}"
    --baseline "${XR_ROOT}/scripts/meta_ownership_baseline.json"
    --max-category AST_PTR_INTO_IR=0 --max-category PTR_ACROSS_GROWTH=0 --max-category CGEN_BORROWED_NAME=0)
foreach(oracle derive_assert_condition_vector derive_assert_equal_vector derive_assert_panics_vector
    derive_equal57_migration derive_nullable58_vectors derive_panic_carrier_vector
    derive_native_projection_vectors derive_sdk_identity derive_sdk_current_identity derive_source_layout_identity)
    add_test(NAME ${oracle} COMMAND "${Python3_EXECUTABLE}" -X utf8 "${CMAKE_CURRENT_LIST_DIR}/../${oracle}.py")
endforeach()
include("${CMAKE_CURRENT_LIST_DIR}/gates.cmake")
