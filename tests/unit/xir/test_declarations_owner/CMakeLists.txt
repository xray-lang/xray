cmake_minimum_required(VERSION 3.21)
project(xray_test_declarations_owner C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Ninja is required")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(PROJECT_SOURCE_DIR "${XR_ROOT}")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(XRAY_PYTHON "${Python3_EXECUTABLE}")
set(XRAY_COMMON_INCLUDES "${XR_ROOT}/src" "${XR_ROOT}/include")
include_directories(${XRAY_COMMON_INCLUDES})
if(MSVC)
    add_compile_options(/W4 /WX /utf-8)
    add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        add_compile_options(/experimental:c11atomics)
    endif()
else()
    add_compile_options(-Wall -Wextra -Werror -pedantic)
endif()
add_custom_target(gen-native-type-defs DEPENDS "${XR_ROOT}/src/shared/xnative_declarations.inc.c")
file(GLOB scripts "${XR_ROOT}/stdlib/*/*.xr")
foreach(path IN LISTS scripts)
    get_filename_component(parent "${path}" DIRECTORY)
    get_filename_component(parent "${parent}" NAME)
    get_filename_component(name "${path}" NAME_WE)
    if(NOT name STREQUAL parent OR parent STREQUAL "types")
        list(REMOVE_ITEM scripts "${path}")
    endif()
endforeach()
file(GLOB_RECURSE generator_inputs "${XR_ROOT}/tools/stdlibgen/*.py" "${XR_ROOT}/stdlib/defs/*.def")
set(embedded "${CMAKE_BINARY_DIR}/generated/stdlib_embedded_sources.inc")
add_custom_command(OUTPUT "${embedded}"
    COMMAND "${Python3_EXECUTABLE}" -X utf8 "${XR_ROOT}/scripts/generate_stdlib_embedded.py"
        --root "${XR_ROOT}" --source-output "${embedded}" ${scripts}
    DEPENDS "${XR_ROOT}/scripts/generate_stdlib_embedded.py" ${generator_inputs}
        "${XR_ROOT}/stdlib/stdlib_boundary.toml" ${scripts} VERBATIM)
add_custom_target(gen-stdlib-embedded-sources DEPENDS "${embedded}")
add_subdirectory("${XR_ROOT}/src/base" base)
add_subdirectory("${XR_ROOT}/src/xir" xir)
enable_testing()
set(old_root "${CMAKE_CURRENT_BINARY_DIR}/old-program27")
file(MAKE_DIRECTORY "${old_root}")
file(READ "${CMAKE_CURRENT_LIST_DIR}/old_program27.json" old_record)
string(JSON old_archive_hash GET "${old_record}" archive sha256)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/old_program27_headers.zip" actual_archive_hash)
if(NOT old_archive_hash STREQUAL actual_archive_hash)
    message(FATAL_ERROR "The preserved old header archive differs from its provenance record")
endif()
string(JSON old_source_hash GET "${old_record}" source sha256)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/old_program27.c" actual_source_hash)
if(NOT old_source_hash STREQUAL actual_source_hash)
    message(FATAL_ERROR "The actual old generated C differs from its provenance record")
endif()
string(JSON old_count LENGTH "${old_record}" archive members)
math(EXPR old_last "${old_count}-1")
set(old_headers)
foreach(index RANGE ${old_last})
    string(JSON path GET "${old_record}" archive members ${index} path)
    list(APPEND old_headers "${old_root}/${path}")
endforeach()
add_custom_command(OUTPUT "${old_root}/extracted.stamp" BYPRODUCTS ${old_headers}
    COMMAND "${CMAKE_COMMAND}" -E tar xf "${CMAKE_CURRENT_LIST_DIR}/old_program27_headers.zip"
    COMMAND "${CMAKE_COMMAND}" -E touch "${old_root}/extracted.stamp"
    DEPENDS "${CMAKE_CURRENT_LIST_DIR}/old_program27_headers.zip" "${CMAKE_CURRENT_LIST_DIR}/old_program27.json"
    WORKING_DIRECTORY "${old_root}" VERBATIM)
add_custom_target(test_test_old_headers DEPENDS "${old_root}/extracted.stamp")
add_library(test_test_old_program27 OBJECT old_program27.c)
add_dependencies(test_test_old_program27 test_test_old_headers)
target_include_directories(test_test_old_program27 BEFORE PRIVATE "${old_root}/src")
