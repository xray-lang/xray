cmake_minimum_required(VERSION 3.25)
project(TestRolesSdkNative C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja" OR NOT WIN32)
    message(FATAL_ERROR "This SDK consumer requires Windows and Ninja")
endif()
if(NOT EXISTS "${XIR_SDK_BUNDLE_DIR}/sdk_manifest.json" OR NOT EXISTS "${XIR_TEST_ROLES_C}")
    message(FATAL_ERROR "A fresh same-source SDK and actual emitted test-role C are required")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(test_test_sdk_native ../test_native.c "${XIR_TEST_ROLES_C}")
target_include_directories(test_test_sdk_native PRIVATE "${XIR_SDK_BUNDLE_DIR}/src")
target_compile_options(test_test_sdk_native PRIVATE /W4 /WX /utf-8)
target_compile_definitions(test_test_sdk_native PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    target_compile_options(test_test_sdk_native PRIVATE /experimental:c11atomics)
endif()
foreach(owner xray_compile_resources xray_xir_admission xray_xir_declarations xray_xir_scalar xray_xir_runtime_host)
    add_library(sdk_${owner} STATIC IMPORTED)
    set_target_properties(sdk_${owner} PROPERTIES IMPORTED_LOCATION "${XIR_SDK_BUNDLE_DIR}/lib/${owner}.lib")
    target_link_libraries(test_test_sdk_native PRIVATE sdk_${owner})
endforeach()
target_link_libraries(test_test_sdk_native PRIVATE kernel32)
enable_testing()
add_test(NAME test_test_sdk_native COMMAND test_test_sdk_native)
