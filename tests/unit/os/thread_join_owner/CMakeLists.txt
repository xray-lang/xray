cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ThreadJoinOwner C)
endif()
if(NOT WIN32 OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This independent gate requires Windows Ninja")
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(test_thread_join_owner test_thread_join.c)
target_compile_definitions(test_thread_join_owner PRIVATE THREAD_JOIN_INJECTED=1)
add_executable(test_thread_join_production test_thread_join.c
    "${XR_ROOT}/src/os/win/thread_win.c")
foreach(target test_thread_join_owner test_thread_join_production)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_include_directories(${target} PRIVATE "${XR_ROOT}/src" "${XR_ROOT}/src/base")
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(${target} PRIVATE /experimental:c11atomics)
    endif()
endforeach()
enable_testing()
add_test(NAME test_thread_join_owner COMMAND test_thread_join_owner)
add_test(NAME test_thread_join_production COMMAND test_thread_join_production)
set_tests_properties(test_thread_join_owner test_thread_join_production PROPERTIES
    LABELS "unit;os;ownership;thread" TIMEOUT 20)
