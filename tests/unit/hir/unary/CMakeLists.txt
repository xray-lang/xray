add_executable(test_hir_unary test_unary.c)
target_link_libraries(test_hir_unary PRIVATE xray_xir_source xray_xir_vm)
add_executable(test_hir_unary_source test_unary_source.c)
if(WIN32)
    target_link_libraries(test_hir_unary_source PRIVATE ${XRAY_SOURCE_SDK_ARCHIVE_TARGETS})
else()
    target_link_libraries(test_hir_unary_source PRIVATE xray_source_api)
endif()
target_include_directories(test_hir_unary_source PRIVATE "${PROJECT_SOURCE_DIR}/include")
if(MSVC)
    set_target_properties(test_hir_unary_source PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
foreach(target test_hir_unary test_hir_unary_source)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
endforeach()
add_test(NAME hir_unary COMMAND test_hir_unary)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/empty-stdlib")
add_test(NAME source_unary_sdk COMMAND test_hir_unary_source "${CMAKE_CURRENT_BINARY_DIR}/empty-stdlib")
set_tests_properties(hir_unary source_unary_sdk PROPERTIES TIMEOUT 60 LABELS "unit;hir;types;ownership;sdk")
if(WIN32)
    add_test(NAME hir_unary_cli COMMAND ${XRAY_PYTHON} -X utf8
        "${CMAKE_CURRENT_SOURCE_DIR}/test_unary_cli.py"
        --root "${PROJECT_SOURCE_DIR}" --binary $<TARGET_FILE:xray>)
    set_tests_properties(hir_unary_cli PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE
        LABELS "unit;hir;types;cli;native")
endif()
