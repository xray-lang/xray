add_executable(test_hir_boolean test_boolean.c)
target_link_libraries(test_hir_boolean PRIVATE xray_xir_source xray_xir_vm)
add_executable(test_hir_boolean_source test_boolean_source.c)
if(WIN32)
    target_link_libraries(test_hir_boolean_source PRIVATE ${XRAY_SOURCE_SDK_ARCHIVE_TARGETS})
else()
    target_link_libraries(test_hir_boolean_source PRIVATE xray_source_api)
endif()
target_include_directories(test_hir_boolean_source PRIVATE "${PROJECT_SOURCE_DIR}/include")
if(MSVC)
    set_target_properties(test_hir_boolean_source PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
foreach(target test_hir_boolean test_hir_boolean_source)
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
endforeach()
