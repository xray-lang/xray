cmake_minimum_required(VERSION 3.24)
project(xray_formatter_owner C)
enable_testing()
if(MSVC)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        add_compile_options(/experimental:c11atomics)
    endif()
endif()
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(FORMAT_SANITIZE)
    add_compile_options(/fsanitize=address /Oy- /clang:-fsanitize=undefined /clang:-fno-sanitize-recover=all)
endif()
add_subdirectory("${XR_ROOT}/src/base" resources)
include("${XR_ROOT}/src/toolchain/xcompiler_producer.cmake")
include("${XR_ROOT}/src/frontend/format/xformatter.cmake")
file(MAKE_DIRECTORY "${XR_ROOT}/build-formatter-owner")
foreach(name test_xfmt_literal test_formatter_strings test_formatter_comments test_fmt_roundtrip)
    add_executable(${name} "${CMAKE_CURRENT_LIST_DIR}/../${name}.c")
    target_link_libraries(${name} PRIVATE xray_frontend_format)
    target_include_directories(${name} PRIVATE "${XR_ROOT}/src/base")
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${XR_ROOT}/build-formatter-owner")
endforeach()
add_executable(test_format_owner test_format_owner.c)
target_link_libraries(test_format_owner PRIVATE xray_frontend_format)
if(MSVC)
    target_compile_options(test_format_owner PRIVATE /W4 /WX /utf-8)
endif()
add_test(NAME test_format_owner COMMAND test_format_owner)
if(FORMAT_SANITIZE)
    execute_process(COMMAND "${CMAKE_C_COMPILER}" /clang:-print-resource-dir
        OUTPUT_VARIABLE resource OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    foreach(name test_xfmt_literal test_formatter_strings test_formatter_comments test_fmt_roundtrip test_format_owner)
        target_link_options(${name} PRIVATE /INCREMENTAL:NO /INCLUDE:__asan_seh_interceptor
            "/WHOLEARCHIVE:${resource}/lib/windows/clang_rt.asan_dynamic_runtime_thunk-x86_64.lib")
        target_link_libraries(${name} PRIVATE "${resource}/lib/windows/clang_rt.asan_dynamic-x86_64.lib")
        set_tests_properties(${name} PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${resource}/lib/windows")
    endforeach()
endif()
