cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DapProcessHandoff C)
endif()
if(NOT CMAKE_GENERATOR STREQUAL "Ninja" OR NOT WIN32)
    message(FATAL_ERROR "This independent gate requires Windows Ninja")
endif()
get_filename_component(HANDOFF_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
foreach(t test_dap_process_handoff test_dap_process_handoff_faults)
    add_executable(${t} test_process_handoff.c
        "${HANDOFF_ROOT}/src/os/win/proc_self_win.c"
        "${HANDOFF_ROOT}/src/os/win/pipe_win.c"
        "${HANDOFF_ROOT}/src/base/xutf8.c")
    set_target_properties(${t} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_include_directories(${t} PRIVATE "${HANDOFF_ROOT}/src")
    target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${t} PRIVATE /W4 /WX /utf-8)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(${t} PRIVATE /experimental:c11atomics)
    endif()
endforeach()
target_sources(test_dap_process_handoff PRIVATE "${HANDOFF_ROOT}/src/os/win/proc_win.c")
target_compile_definitions(test_dap_process_handoff_faults PRIVATE XDAP_PROCESS_INJECTED=1)
enable_testing()
add_test(NAME test_dap_process_handoff COMMAND test_dap_process_handoff)
add_test(NAME test_dap_process_handoff_faults COMMAND test_dap_process_handoff_faults)
add_test(NAME test_dap_process_handoff_resources COMMAND test_dap_process_handoff_faults --resources)
set_tests_properties(test_dap_process_handoff test_dap_process_handoff_faults
    PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE PROCESSORS 1 LABELS "dap;process;ownership")
set_tests_properties(test_dap_process_handoff_resources
    PROPERTIES TIMEOUT 300 RUN_SERIAL TRUE PROCESSORS 1 LABELS "dap;process;ownership;resources;fault-injection")
if(COMMAND xray_stage_windows_asan_runtime)
    xray_stage_windows_asan_runtime("${CMAKE_CURRENT_BINARY_DIR}")
    xray_prepend_windows_asan_runtime_to_directory_tests()
endif()
