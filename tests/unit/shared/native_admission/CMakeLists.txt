cmake_minimum_required(VERSION 3.21)
project(xray_native_admission_tests LANGUAGES C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Native admission tests require Ninja")
endif()
get_filename_component(NATIVE_ADMISSION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(test_native_declaration_admission test_native_admission.c
    "${NATIVE_ADMISSION_ROOT}/src/shared/xnative_declaration.c")
target_include_directories(test_native_declaration_admission PRIVATE "${NATIVE_ADMISSION_ROOT}/src")
target_compile_features(test_native_declaration_admission PRIVATE c_std_11)
if(MSVC)
    target_compile_options(test_native_declaration_admission PRIVATE /W4 /WX /utf-8)
    target_compile_definitions(test_native_declaration_admission PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(test_native_declaration_admission PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME test_native_declaration_admission COMMAND test_native_declaration_admission)
set_tests_properties(test_native_declaration_admission PROPERTIES LABELS "unit;compiler;budget" TIMEOUT 60)
