cmake_minimum_required(VERSION 3.21)
project(xray_bits_wide_tests LANGUAGES C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Wide multiply tests require Ninja")
endif()
get_filename_component(XR_BITS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
enable_testing()
set(bits_targets test_bits_wide_c11)
add_executable(test_bits_wide_c11 test_bits_wide.c)
if(NOT MSVC)
    add_executable(test_bits_wide_fallback test_bits_wide.c)
    target_compile_options(test_bits_wide_fallback PRIVATE -U__SIZEOF_INT128__)
    list(APPEND bits_targets test_bits_wide_fallback)
endif()
foreach(target IN LISTS bits_targets)
    target_include_directories(${target} PRIVATE "${XR_BITS_ROOT}/src")
    target_compile_features(${target} PRIVATE c_std_11)
    set_target_properties(${target} PROPERTIES C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES LABELS "unit;numeric;portability" TIMEOUT 30)
endforeach()
if(CMAKE_C_COMPILER_ID STREQUAL "MSVC" AND TARGET xray)
    find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
    add_test(NAME test_generated_c11_msvc COMMAND ${Python3_EXECUTABLE}
        "${XR_BITS_ROOT}/tests/cli/test_generated_c11_msvc.py"
        --xray $<TARGET_FILE:xray> --cc "${CMAKE_C_COMPILER}" --root "${XR_BITS_ROOT}")
    set_tests_properties(test_generated_c11_msvc PROPERTIES LABELS "unit;generated-c;portability" TIMEOUT 150)
endif()
