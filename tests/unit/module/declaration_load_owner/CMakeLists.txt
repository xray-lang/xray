cmake_minimum_required(VERSION 3.21)
project(xray_declaration_load_owner_tests LANGUAGES C)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Declaration load tests require Ninja")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_library(declaration_load_sources STATIC "${XR_ROOT}/src/base/xio_policy.c"
    "${XR_ROOT}/src/base/xtoml.c" "${XR_ROOT}/src/os/win/file_read_win.c"
    "${XR_ROOT}/src/module/xdeclaration_manifest.c" "${XR_ROOT}/src/module/xdeclaration_load.c")
add_executable(test_declaration_file_owner "${XR_ROOT}/tests/unit/base/test_declaration_file_windows.c")
add_executable(test_declaration_schema_owner "${XR_ROOT}/tests/unit/module/test_declaration_manifest.c")
foreach(target declaration_load_sources test_declaration_file_owner test_declaration_schema_owner)
    target_include_directories(${target} PRIVATE "${XR_ROOT}/src")
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /WX)
        if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
            target_compile_options(${target} PRIVATE /experimental:c11atomics)
        endif()
    endif()
endforeach()
target_link_libraries(test_declaration_file_owner PRIVATE declaration_load_sources)
target_link_libraries(test_declaration_schema_owner PRIVATE declaration_load_sources)
add_test(NAME test_declaration_file_owner COMMAND ${Python3_EXECUTABLE}
    "${XR_ROOT}/tests/unit/base/test_declaration_file_windows.py" $<TARGET_FILE:test_declaration_file_owner>)
add_test(NAME test_declaration_schema_owner COMMAND test_declaration_schema_owner)
set_tests_properties(test_declaration_file_owner test_declaration_schema_owner PROPERTIES
    LABELS "unit;metadata;ownership;budget" TIMEOUT 120)
